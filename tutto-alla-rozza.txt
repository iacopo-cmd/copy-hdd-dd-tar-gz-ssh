    [[ "$NOME_BASE" =~ ^[A-Za-z0-9._-]+$ ]] || echo "NOME_BASE: solo lettere, numeri . _ -"
    [[ "$CARTELLA" == /* && "$CARTELLA" != *"'"* ]] || echo "CARTELLA: deve iniziare con / (niente apici)"
    if [[ "$DD_BLOCCO_MB" =~ ^[0-9]+$ ]]; then
        (( DD_BLOCCO_MB >= 4 )) || echo "DD_BLOCCO_MB: minimo 4 (ora: $DD_BLOCCO_MB)"

# CHIAVE HOST della destinazione: la prima volta la prendo con ssh-keyscan
# (come StrictHostKeyChecking=accept-new, ma va anche con OpenSSH vecchi, es. Debian 7-9)
HOST_ID="$IP"; [ "$PORTA" != 22 ] && HOST_ID="[$IP]:$PORTA"
host_noto() {
    # prende le chiavi host che MANCANO, tipo per tipo (rsa, ecdsa, ed25519):
    # gli ssh-keyscan vecchi senza -t prendono solo rsa, ma ssh vuole ecdsa.
    # Un tipo gia' noto NON viene toccato: se la chiave e' cambiata, ssh se ne accorge.
    local T P NOTI AGGIUNTE=""
    mkdir -p /root/.ssh && touch /root/.ssh/known_hosts && chmod 600 /root/.ssh/known_hosts
    NOTI=$(ssh-keygen -F "$HOST_ID" 2>/dev/null | awk '!/^#/ {print $2}')
    for T in rsa ecdsa ed25519; do
        case "$T" in rsa) P=ssh-rsa ;; ecdsa) P=ecdsa-sha2 ;; ed25519) P=ssh-ed25519 ;; esac
        grep -q "^$P" <<< "$NOTI" && continue
        ssh-keyscan -p "$PORTA" -T 15 -t "$T" "$IP" 2>/dev/null | grep -v '^#' >> /root/.ssh/known_hosts &&
            AGGIUNTE+="$T "
    done
    [ -n "$AGGIUNTE" ] && echo "Chiavi host di $HOST_ID salvate in /root/.ssh/known_hosts: $AGGIUNTE"
    ssh-keygen -F "$HOST_ID" >/dev/null 2>&1
}

    fi

fi

# ESITO DI UNA CHIAMATA SSH
# sshpass 5 = password rifiutata -> STOP subito (riprovare farebbe scattare fail2ban)
# sshpass 6 = chiave host da confermare -> STOP
# ssh 255 / sshpass 3,4 = connessione caduta o impossibile -> si riprova
TENTATIVI=0
caduta() {   # $1 = codice di uscita; ritorna 0 se si deve riprovare
    case "$1" in
        5) echo "ERRORE: password rifiutata da $REMOTE. Mi fermo (niente retry, rischio ban)." >&2
           if [ "$PASS_ORIGINE" = "file" ]; then
               rm -f "$PASS_FILE"
               echo "Cancellata la password salvata: al prossimo avvio la richiedo." >&2
           fi
           unset SSHPASS; exit 1 ;;
        6) echo "ERRORE: chiave host di $IP da confermare. Mi fermo." >&2
           unset SSHPASS; exit 1 ;;
        255|3|4)
           # chiave host mai vista (destinazione irraggiungibile al primo giro): riprovo a prenderla
           host_noto >&2
           # 255 non vuol dire sempre "rete caduta": guardo il motivo vero
           MOTIVO=$($SSH "$REMOTE" true 2>&1 </dev/null)
           if grep -q 'IDENTIFICATION HAS CHANGED' <<< "$MOTIVO"; then
               echo "ERRORE: la chiave host di $HOST_ID e' CAMBIATA (destinazione reinstallata?). Mi fermo." >&2
               echo "Se sei sicuro che e' la macchina giusta: ssh-keygen -R '$HOST_ID' e rilancia." >&2
               unset SSHPASS; exit 1
           fi
           if [ "$PASS_ORIGINE" = "chiave" ] && grep -q 'Permission denied' <<< "$MOTIVO"; then
               echo "ERRORE: la destinazione rifiuta la chiave $CHIAVE_USATA. Mi fermo: rilancia chiavi.sh." >&2
               exit 1
           fi
           TENTATIVI=$((TENTATIVI+1))
           if (( MAX_TENTATIVI > 0 && TENTATIVI > MAX_TENTATIVI )); then
               echo "ERRORE: superati $MAX_TENTATIVI tentativi. Mi fermo: rilancia per riprendere." >&2
               unset SSHPASS; exit 1
           fi
           echo "$(date '+%H:%M:%S') CONNESSIONE CADUTA (codice $1), tentativo $TENTATIVI: riprovo tra $ATTESA_RIPROVA s..." >&2
           sleep "$ATTESA_RIPROVA"
           return 0 ;;
    esac
    return 1
}

# SSH CHE RIPROVA DA SOLO SE CADE LA CONNESSIONE (la password la rimanda sshpass)
# rssh "comando" [dati da mandare su stdin]
rssh() {
    local RC
    while true; do
        if [ $# -ge 2 ]; then
            printf '%s\n' "$2" | $SSH "$REMOTE" "$1"; RC=$?
        else
            $SSH "$REMOTE" "$1" </dev/null; RC=$?
        fi
        caduta "$RC" && continue
        TENTATIVI=0
        return "$RC"
    done
}

# Misuratore del flusso dati
meter() {
perl -e '
use Time::HiRes qw(time);

my ($label,$total,$ogni)=@ARGV;
my $start=time;
my $last=$start;
my $bytes=0;
my $buf;
$|=1;

while (1) {
    my $r=sysread(STDIN,$buf,1048576);
    die "Errore lettura\n" unless defined $r;
    last if $r==0;

    my $off=0;
    while ($off<$r) {
        my $w=syswrite(STDOUT,$buf,$r-$off,$off);
        die "Errore scrittura\n" unless defined $w && $w>0;
        $off+=$w;
    }

    $bytes+=$r;
    my $now=time;

    if ($now-$last>=$ogni) {
        my $speed=$bytes/1048576/($now-$start);

        if ($total>0) {
            printf STDERR
            "[%s] %.1f / %.1f MiB (%.1f%%) - %.2f MiB/s\n",
            $label,$bytes/1048576,$total/1048576,
            100*$bytes/$total,$speed;
        } else {
            printf STDERR
            "[%s] %.1f MiB prodotti - %.2f MiB/s\n",
            $label,$bytes/1048576,$speed;
        }

        $last=$now;
    }
}

my $elapsed=time-$start;
my $speed=$elapsed>0 ? $bytes/1048576/$elapsed : 0;

printf STDERR
"[%s] COMPLETATO: %.1f MiB in %.1f secondi - %.2f MiB/s\n",
$label,$bytes/1048576,$elapsed,$speed;
' "$1" "$2" "$MISURA_OGNI"
}

# DIMENSIONE DISCO
SIZE=$(blockdev --getsize64 "$DISK") || exit 1

# ---------------- CHECK DEL BLOCCO DI LETTURA ----------------
# Prova piu' grandezze di blocco leggendo CHECK_MB per ognuna (solo lettura, senza cache)
# e prende la piu' veloce. Il valore del config vince se l'altra non e' almeno il 5% meglio.
# Se il check fallisce si usa DD_BLOCCO_MB del config.
SCELTA_BLOCCO="config (AUTO_BLOCCO=no)"
CHECK_TXT=""

check_blocco() {
    local DEV NOMEDEV TIPO="disco" OPT EXT C I OFF CNT T0 T1 MS VEL
    local DIRECT="iflag=direct" CAND=() OKC=() OKV=() VEL_CONF=0 MIGLIORE=0 VEL_MIGL=0
    DEV=$(readlink -f "$DISK")
    NOMEDEV=$(basename "$DEV")

    CAND=(4 8 16 "$DD_BLOCCO_MB")

    # device-mapper (LVM, LUKS...): blocchi piu' grandi
    if [ -d "/sys/class/block/$NOMEDEV/dm" ]; then
        TIPO="device-mapper"
        CAND+=(32 64)
        if command -v lvs >/dev/null 2>&1 &&
           EXT=$(lvs --noheadings --nosuffix --units m -o vg_extent_size "$DISK" 2>/dev/null); then
            EXT=$(tr -d ' ' <<< "$EXT"); EXT=${EXT%%.*}
            if [[ "$EXT" =~ ^[0-9]+$ ]] && (( EXT > 0 )); then
                TIPO="LVM (extent $EXT MB)"; CAND+=("$EXT")
            fi
        fi
    fi

    # grandezza suggerita dal disco (byte)
    OPT=$(cat "/sys/class/block/$NOMEDEV/queue/optimal_io_size" 2>/dev/null)
    if [[ "$OPT" =~ ^[0-9]+$ ]] && (( OPT >= 4194304 && OPT % 1048576 == 0 )); then
        CAND+=($((OPT/1048576)))
    fi

    # tieni solo: >= 4, divide SPLIT_MB, <= CHECK_MB; senza doppioni
    mapfile -t CAND < <(for C in "${CAND[@]}"; do
        (( C >= 4 && SPLIT_MB % C == 0 && C <= CHECK_MB )) && echo "$C"; done | sort -n | uniq)

    echo "Check blocco su $DISK ($TIPO): provo ${CAND[*]} MB, $CHECK_MB MB ciascuno..."
    (( ${#CAND[@]} == 0 )) && return 1
    (( SIZE < CHECK_MB*1048576 )) && { echo "  disco piu' piccolo di CHECK_MB"; return 1; }

    # lettura diretta (senza cache) se il disco la supporta
    dd if="$DISK" of=/dev/null bs=4M count=1 $DIRECT 2>/dev/null || {
        DIRECT=""; echo "  (lettura diretta non supportata: uso zone diverse del disco)"; }

    I=0
    for C in "${CAND[@]}"; do
        # zona diversa per ogni prova, cosi' la cache non falsa i numeri
        OFF=$(( I*CHECK_MB ))
        (( (OFF+CHECK_MB)*1048576 > SIZE )) && OFF=0
        CNT=$(( CHECK_MB/C ))
        T0=$(date +%s%N)
        if dd if="$DISK" of=/dev/null bs="${C}M" skip=$((OFF/C)) count="$CNT" $DIRECT 2>/dev/null; then
            T1=$(date +%s%N)
            MS=$(( (T1-T0)/1000000 )); (( MS < 1 )) && MS=1
            VEL=$(( CNT*C*1000/MS ))
            printf "  %3d MB -> %5d MB/s\n" "$C" "$VEL"
            CHECK_TXT+="CHECK_${C}MB=${VEL} MB/s"$'\n'
            OKC+=("$C"); OKV+=("$VEL")
            (( C == DD_BLOCCO_MB )) && VEL_CONF=$VEL
            (( VEL > VEL_MIGL )) && { VEL_MIGL=$VEL; MIGLIORE=$C; }
        else
            echo "  $C MB -> errore di lettura, scartato"
        fi
        I=$((I+1))
    done

    (( ${#OKC[@]} == 0 )) && return 1

    if (( VEL_CONF > 0 && MIGLIORE != DD_BLOCCO_MB && VEL_MIGL*100 < VEL_CONF*105 )); then
        SCELTA_BLOCCO="config ($DD_BLOCCO_MB MB: $MIGLIORE MB solo ${VEL_MIGL} vs ${VEL_CONF} MB/s, meno del 5% meglio)"
    elif (( MIGLIORE == DD_BLOCCO_MB )); then
        SCELTA_BLOCCO="check (il config $DD_BLOCCO_MB MB e' il piu' veloce)"
    else
        SCELTA_BLOCCO="check ($MIGLIORE MB a ${VEL_MIGL} MB/s, il config $DD_BLOCCO_MB MB faceva ${VEL_CONF} MB/s)"
        DD_BLOCCO_MB=$MIGLIORE
    fi
    return 0
}

if [ "$AUTO_BLOCCO" = "si" ]; then

        echo "Check fallito: uso il blocco del config, $DD_BLOCCO_MB MB."
    fi
fi

#!/bin/bash
# Crea una chiave SSH su QUESTA macchina (la sorgente) e la installa sulla destinazione.

host_noto() {

            AGGIUNTE+="$T "
