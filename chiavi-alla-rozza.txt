# Crea /root/backup-disco/chiavi.sh e lo lancia: chiave SSH + installazione sulla destinazione.
# La password della destinazione la chiede una volta sola.
D=/root/backup-disco; mkdir -p "$D" && chmod 700 "$D"
cat > "$D/chiavi.sh" <<'FINE_ALLA_ROZZA_1793792394'
#!/bin/bash
# Crea una chiave SSH su QUESTA macchina (la sorgente) e la installa sulla destinazione.
# La password della destinazione la chiede ssh UNA volta sola, poi non serve piu'.
# Dopo, backup-disco.sh usa la chiave da solo (CHIAVE_SSH=auto in config.txt).
#
# PERCHE' LA CHIAVE AUTOMATICA
# Questo strumento serve quando le cose vanno male: un crash, un disco che cede, un sistema
# da spostare di corsa. In quei momenti hai fretta, sei agitato e non hai voglia di scrivere:
# e' li' che sbagli una password, un IP, un comando. Con la chiave scrivi la password UNA volta,
# poi il backup va da solo per ore, riprova se cade la rete e riparte dopo un riavvio
# senza chiederti niente. Meno cose da scrivere = meno errori quando conta.

D="$(dirname "$(readlink -f "$0")")"
CONF="$D/config.txt"
CHIAVE=/root/.ssh/backup-disco_ed25519

# valori proposti: da config.txt se c'e'
leggi() { [ -f "$CONF" ] && sed -nE "s/^$1=([^[:space:]#]*).*/\1/p" "$CONF" | tail -1; }
IP=$(leggi IP); PORTA=$(leggi PORTA); UTENTE=$(leggi UTENTE)
[ "$IP" = "0.0.0.0" ] && IP=""
[ -z "$PORTA" ] && PORTA=22
[ -z "$UTENTE" ] && UTENTE=root

echo "=== Chiave SSH per il backup (INVIO = tengo il valore tra [ ]) ==="
read -r -p "IP destinazione [$IP]: " R;  [ -n "$R" ] && IP="$R"
[ -z "$IP" ] && { echo "ERRORE: IP vuoto"; exit 1; }
read -r -p "Porta [$PORTA]: " R;         [ -n "$R" ] && PORTA="$R"
read -r -p "Utente [$UTENTE]: " R;       [ -n "$R" ] && UTENTE="$R"

command -v ssh-keygen >/dev/null 2>&1 || { echo "ERRORE: manca ssh-keygen (apt install openssh-client)"; exit 1; }

# 1. CHIAVE (senza passphrase: serve per andare da solo)
mkdir -p /root/.ssh && chmod 700 /root/.ssh
if [ -f "$CHIAVE" ]; then
    echo "Chiave gia' presente, la riuso: $CHIAVE"
else
    ssh-keygen -q -t ed25519 -N "" -C "backup-disco@$(hostname)" -f "$CHIAVE" || { echo "ERRORE: chiave non creata"; exit 1; }
    echo "Chiave creata: $CHIAVE"
fi
PUB=$(cat "$CHIAVE.pub")

# 2. INSTALLAZIONE SULLA DESTINAZIONE (qui ssh chiede la password, una volta)
echo ""
echo "Ora scrivi la password di $UTENTE@$IP (una volta sola):"
if ! ssh -p "$PORTA" -o StrictHostKeyChecking=accept-new -o PubkeyAuthentication=no \
        -o PreferredAuthentications=password,keyboard-interactive -o NumberOfPasswordPrompts=2 \
        "$UTENTE@$IP" "umask 077; mkdir -p ~/.ssh && touch ~/.ssh/authorized_keys &&
        { grep -qxF '$PUB' ~/.ssh/authorized_keys || echo '$PUB' >> ~/.ssh/authorized_keys; } &&
        chmod 700 ~/.ssh && chmod 600 ~/.ssh/authorized_keys &&
        { command -v restorecon >/dev/null 2>&1 && restorecon -R ~/.ssh; true; } && echo INSTALLATA"
then
    echo ""
    echo "ERRORE: non sono riuscito a installare la chiave."
    echo " - password sbagliata? riprova"
    echo " - se dice 'Permission denied (publickey)': la destinazione NON accetta password."
    echo "   Allora entra dalla console del provider e incolla questa riga in ~/.ssh/authorized_keys:"
    echo "   $PUB"
    exit 1
fi

# 3. PROVA: deve entrare SENZA password
echo ""
if ssh -i "$CHIAVE" -p "$PORTA" -o IdentitiesOnly=yes -o BatchMode=yes -o ConnectTimeout=20 \
       "$UTENTE@$IP" "echo OK" >/dev/null 2>&1; then
    echo "PROVA OK: $UTENTE@$IP entra con la chiave, senza password."
else
    echo "ATTENZIONE: chiave installata ma la prova senza password fallisce."
    echo "Sulla destinazione controlla in /etc/ssh/sshd_config: PubkeyAuthentication yes"
    exit 1
fi

# 4. CONFIG: scrivo IP, porta, utente e CHIAVE_SSH (backup del vecchio)
if [ -f "$CONF" ]; then
    cp -p "$CONF" "$CONF.vecchio-$(date +%Y%m%d-%H%M%S)"
    for V in IP PORTA UTENTE CHIAVE_SSH; do
        case $V in IP) X="$IP";; PORTA) X="$PORTA";; UTENTE) X="$UTENTE";; CHIAVE_SSH) X="$CHIAVE";; esac
        if grep -q "^$V=" "$CONF"; then
            sed -i -E "s|^$V=[^[:space:]#]*|$V=$X|" "$CONF"
        else
            echo "$V=$X" >> "$CONF"
        fi
    done
    echo "Aggiornato $CONF (IP, PORTA, UTENTE, CHIAVE_SSH). Il vecchio e' in config.txt.vecchio-*"
fi
echo ""
echo "Fatto. Ora lancia: $D/backup-disco.sh (meglio dentro tmux)"
FINE_ALLA_ROZZA_1793792394
chmod 700 "$D/chiavi.sh"
bash -n "$D/chiavi.sh" && echo "OK: $D/chiavi.sh" || echo "ERRORE: chiavi.sh incollato male, ricopia"
unset D
 set -o history
/root/backup-disco/chiavi.sh
